## pc/receiver.py

```python
"""FPG1 UART receiver. Pure stdlib parsing/PGM output; pyserial for a real port.

CRC-16/CCITT-FALSE: polynomial 0x1021, init 0xffff, no reflection/xorout.
CRC covers header[4:] and the wire payload and is sent little endian. Version 1
has a 32-byte header; versions 2/3 have a 72-byte header. Version 3 transports
packed binary Sobel pixels or only ROI target statistics. Coordinates
refer to the sensor (the cropped image origin is sensor (1,1)). The legacy
bright-pixel centroid remains separate from the version 2 target centroid.
"""
import argparse
import binascii
import csv
import json
import math
from pathlib import Path
import struct
import time

MAGIC = b'FPG1'
HEADER = struct.Struct('<4sBBHHHIIIII')
TARGET_EXTENSION = struct.Struct('<BBHHHHHHHHIIIIIH')
KEYS = ('version', 'mode', 'width', 'height', 'threshold', 'frame_id',
        'payload_length', 'bright_count', 'sum_x', 'sum_y')
MAX_PIXELS = 320 * 240
BOX_KEYS = ('x_min', 'y_min', 'x_max', 'y_max')
TARGET_CSV_KEYS = ('config_valid', 'valid', 'dark', 'threshold',
                   'roi_x_min', 'roi_y_min', 'roi_x_max', 'roi_y_max',
                   'area', 'sum_x', 'sum_y', 'min_area', 'max_area',
                   'bbox_x_min', 'bbox_y_min', 'bbox_x_max', 'bbox_y_max',
                   'centroid_x', 'centroid_y')
CSV_FIELDS = (*KEYS, 'centroid_x', 'centroid_y', 'image_file',
              *('target_' + key for key in TARGET_CSV_KEYS), 'pixel_format')
MONO1_BYTES = tuple(bytes(255 if value & (1 << bit) else 0 for bit in range(8))
                    for value in range(256))


def _sum_bounds(area, low, high, capacity):
    """Possible axis sums for distinct pixels inside a rectangular box."""
    whole, remainder = divmod(area, capacity)
    minimum = area * low + capacity * whole * (whole - 1) // 2 + whole * remainder
    maximum = area * (low + high) - minimum
    # Both bounding-box endpoints must actually occur in the selected pixels.
    if low != high:
        minimum = max(minimum, (area - 1) * low + high)
        maximum = min(maximum, (area - 1) * high + low)
    return minimum, maximum


def _target_fields(values, width, height):
    (flags, threshold, rx0, ry0, rx1, ry1, bx0, by0, bx1, by1,
     area, sx, sy, minimum, maximum, reserved) = values
    if flags & ~0x07 or reserved:
        return False
    valid, dark, config_valid = bool(flags & 1), bool(flags & 2), bool(flags & 4)
    expected_config = (1 <= rx0 <= rx1 <= width and 1 <= ry0 <= ry1 <= height
                       and 1 <= minimum <= maximum <= width * height)
    if config_valid != expected_config:
        return False
    if valid != (config_valid and area > 0 and minimum <= area <= maximum):
        return False
    if not config_valid or not area:
        if any((area, sx, sy, bx0, by0, bx1, by1)):
            return False
    else:
        if not (rx0 <= bx0 <= bx1 <= rx1 and ry0 <= by0 <= by1 <= ry1):
            return False
        if area > (bx1 - bx0 + 1) * (by1 - by0 + 1):
            return False
        if area == 1 and (bx0 != bx1 or by0 != by1):
            return False
        sx_min, sx_max = _sum_bounds(area, bx0, bx1, by1 - by0 + 1)
        sy_min, sy_max = _sum_bounds(area, by0, by1, bx1 - bx0 + 1)
        if not (sx_min <= sx <= sx_max and sy_min <= sy <= sy_max):
            return False
    return dict(config_valid=config_valid, valid=valid, dark=dark,
                threshold=threshold, roi=dict(zip(BOX_KEYS, (rx0, ry0, rx1, ry1))),
                area=area, sum_x=sx, sum_y=sy, min_area=minimum, max_area=maximum,
                bbox=dict(zip(BOX_KEYS, (bx0, by0, bx1, by1))) if valid else None,
                centroid_x=sx // area if valid else None,
                centroid_y=sy // area if valid else None)


class FrameParser:
    """Incremental bounded parser; feed returns a list of validated frame dicts.

    Normal feeds always wait for the leading packet's declared length, so image
    pixels containing a valid packet cannot change parsing across read chunks.
    Only an explicit stream boundary may recover a later CRC-verified packet
    from an incomplete transfer. Call finish afterward to discard its tail.
    """
    def __init__(self):
        self.buffer = bytearray()
        self.rejected = 0

    def _candidate(self, pos):
        if len(self.buffer) - pos < HEADER.size:
            return None, None
        values = HEADER.unpack_from(self.buffer, pos)
        f = dict(zip(KEYS, values[1:]))
        w, h, n = f['width'], f['height'], f['payload_length']
        pixel_count = w * h
        if (values[0] != MAGIC or f['version'] not in (1, 2, 3) or f['mode'] not in (0, 1)
                or not (1 <= w <= 318 and 1 <= h <= 238)
                or n > MAX_PIXELS or f['threshold'] > 2047
                or f['bright_count'] > pixel_count
                or f['sum_x'] > f['bright_count'] * 319
                or f['sum_y'] > f['bright_count'] * 239):
            return False, None
        if f['version'] != 3 and n != pixel_count:
            return False, None
        header_size = HEADER.size
        pixel_format = 'gray8'
        if f['version'] in (2, 3):
            header_size += TARGET_EXTENSION.size
            if len(self.buffer) - pos < header_size:
                return None, None
            extension = TARGET_EXTENSION.unpack_from(self.buffer, pos + HEADER.size)
            has_target = True
            if f['version'] == 3:
                transport = self.buffer[pos + 70]
                if transport not in (1, 5, 6) or self.buffer[pos + 71]:
                    return False, None
                pixel_format = 'none' if transport == 6 else 'mono1'
                if pixel_format == 'mono1':
                    if f['mode'] != 1 or n != (pixel_count + 7) // 8:
                        return False, None
                elif n != 0:
                    return False, None
                has_target = transport != 1
                if not has_target and any(self.buffer[pos + 32:pos + 70]):
                    return False, None
                # The v2 reserved word is transport_flags plus reserved in v3.
                extension = (*extension[:-1], 0)
            if has_target:
                target = _target_fields(extension, w, h)
                if target is False:
                    return False, None
                f['target'] = target
        end = pos + header_size + n + 2
        if len(self.buffer) < end:
            return None, None
        expected, = struct.unpack_from('<H', self.buffer, end - 2)
        actual = binascii.crc_hqx(self.buffer[pos + 4:end - 2], 0xffff)
        if actual != expected:
            return False, None
        payload = bytes(self.buffer[pos + header_size:end - 2])
        if pixel_format == 'mono1':
            remaining = pixel_count % 8
            if remaining and payload[-1] & (0xff << remaining):
                return False, None
            payload = b''.join(MONO1_BYTES[value] for value in payload)[:pixel_count]
        f['pixels'] = payload
        f['pixel_format'] = pixel_format
        count = f['bright_count']
        f['centroid_x'] = f['sum_x'] // count if count else None
        f['centroid_y'] = f['sum_y'] // count if count else None
        return f, end

    def feed(self, chunk, *, boundary=False):
        self.buffer.extend(chunk)
        result = []
        while True:
            pos = self.buffer.find(MAGIC)
            if pos < 0:
                # Preserve possible fragmented magic, never unbounded noise.
                self.buffer[:] = self.buffer[-3:]
                break
            if pos:
                del self.buffer[:pos]
            frame, end = self._candidate(0)
            if frame is False:
                self.rejected += 1
                del self.buffer[0]
                continue
            if frame is None:
                if not boundary:
                    break
                # EOF or a caller-confirmed idle gap ends the old transfer.
                # Mere read fragmentation must never trigger this recovery.
                later = self.buffer.find(MAGIC, 4)
                recovered = False
                while later >= 0:
                    candidate, _ = self._candidate(later)
                    if candidate:
                        del self.buffer[:later]
                        self.rejected += 1
                        recovered = True
                        break
                    later = self.buffer.find(MAGIC, later + 1)
                if recovered:
                    continue
                break
            result.append(frame)
            del self.buffer[:end]
        return result

    def finish(self):
        """Discard ended-transfer leftovers; report full-magic truncation.

        A shorter magic suffix may simply be noise and is not enough evidence
        to classify a transfer as truncated. Call after a real stream EOF or
        an explicit idle boundary ends the transfer, not a single read timeout.
        """
        incomplete = self.buffer.startswith(MAGIC)
        if incomplete:
            self.rejected += 1
        self.buffer.clear()
        return incomplete


def display_geometry(frame):
    """Return ROI/target overlay positions in cropped-image coordinates."""
    target = frame.get('target')
    if target is None:
        return None

    def corners(box):
        return ((box['x_min'] - 1, box['y_min'] - 1),
                (box['x_max'] - 1, box['y_max'] - 1))

    return dict(roi=corners(target['roi']) if target['config_valid'] else None,
                bbox=corners(target['bbox']) if target['valid'] else None,
                centroid=(target['centroid_x'] - 1, target['centroid_y'] - 1)
                if target['valid'] else None,
                status=('Target' if target['valid'] else 'No target')
                if target['config_valid'] else 'Invalid target configuration')


def draw_target_overlay(picture, frame, cv2):
    """Draw only on a display copy; raw PGM and frame pixels stay unchanged."""
    geometry = display_geometry(frame)
    if geometry is None:
        return picture
    display = cv2.cvtColor(picture, cv2.COLOR_GRAY2BGR)
    if geometry['roi']:
        cv2.rectangle(display, *geometry['roi'], (0, 200, 255), 1)
    if geometry['bbox']:
        cv2.rectangle(display, *geometry['bbox'], (0, 255, 0), 1)
        cv2.circle(display, geometry['centroid'], 3, (0, 0, 255), -1)
    label = geometry['status']
    if frame.get('pixel_format') == 'none':
        label = 'Features only - ' + label
    if geometry['centroid']:
        target = frame['target']
        label += f" sensor=({target['centroid_x']},{target['centroid_y']})"
    cv2.putText(display, label, (4, 16), cv2.FONT_HERSHEY_SIMPLEX, 0.4,
                (0, 255, 0) if geometry['bbox'] else (0, 0, 255), 1)
    return display


def display_frame(frame, cv2, np):
    """Create the current frame's canvas; result-only packets never reuse pixels."""
    if frame.get('pixel_format') == 'none':
        picture = np.zeros((frame['height'], frame['width']), dtype=np.uint8)
    else:
        picture = np.frombuffer(frame['pixels'], dtype=np.uint8).reshape(
            frame['height'], frame['width'])
    return draw_target_overlay(picture, frame, cv2)


class PreviewCache:
    """Optional display-only background; metadata and saved pixels stay current.

    A reset, repeated ID, or incompatible image settings invalidates the held
    image. The wire format has no source/session ID, so an undetectable camera
    change cannot be inferred here; the sender must supply a fresh preview.
    """
    def __init__(self):
        self.pixels = None
        self.preview_id = None
        self.last_id = None
        self.settings = None

    def render(self, frame, cv2, np):
        settings = (frame['width'], frame['height'], frame['mode'], frame['threshold'])
        if (settings != self.settings or
                (self.last_id is not None and frame['frame_id'] <= self.last_id)):
            self.pixels = None
            self.preview_id = None
        self.settings = settings
        self.last_id = frame['frame_id']
        if frame.get('pixel_format') != 'none':
            # Keep immutable, undecorated pixels, never an OpenCV display array.
            self.pixels = bytes(frame['pixels'])
            self.preview_id = frame['frame_id']
            return display_frame(frame, cv2, np)
        if self.pixels is None:
            return display_frame(frame, cv2, np)
        picture = np.frombuffer(self.pixels, dtype=np.uint8).reshape(
            frame['height'], frame['width'])
        display = draw_target_overlay(picture, frame, cv2)
        # Separate lines remain readable at the normal 318-pixel image width.
        for y, label in ((34, f'Preview frame={self.preview_id} (held)'),
                         (50, f'Target frame={frame["frame_id"]}')):
            cv2.putText(display, label, (4, y), cv2.FONT_HERSHEY_SIMPLEX,
                        0.4, (0, 200, 255), 1)
        return display


def read_serial_chunk(stream):
    """Wait for one byte, then drain only bytes already available (at most 4096).

    Reading 4096 directly with a timeout batches short features-only packets
    until the timeout expires, even though each packet is already complete.
    """
    first = stream.read(1)
    if not first:
        return first
    available = min(stream.in_waiting, 4095)
    return first + stream.read(available) if available else first


def _csv_path(out):
    """Keep old or foreign CSV schemas intact instead of appending shifted rows."""
    suffix = 0
    while True:
        name = 'metadata.csv' if suffix == 0 else (
            'metadata_v3.csv' if suffix == 1 else f'metadata_v3_{suffix - 1}.csv')
        path = out / name
        if not path.exists() or path.stat().st_size == 0:
            return path
        try:
            with path.open(newline='', encoding='utf-8-sig') as handle:
                if next(csv.reader(handle), None) == list(CSV_FIELDS):
                    return path
        except (UnicodeError, csv.Error):
            # A foreign or damaged schema is not safe to append to.
            pass
        suffix += 1


def _csv_row(meta):
    row = {key: value for key, value in meta.items() if key != 'target'}
    target = meta.get('target')
    if target is not None:
        for key in TARGET_CSV_KEYS:
            if key.startswith('roi_'):
                value = target['roi'][key[4:]]
            elif key.startswith('bbox_'):
                value = target['bbox'][key[5:]] if target['bbox'] else None
            else:
                value = target[key]
            row['target_' + key] = value
    return row


def save_frame(frame, out):
    """Write metadata and, when present, a lossless expanded-pixel PGM.

    Return the image path, or the JSON path for a result-only packet.
    """
    out = Path(out)
    out.mkdir(parents=True, exist_ok=True)
    base = f"frame_{frame['frame_id']:010d}"
    image_present = frame.get('pixel_format', 'gray8') != 'none'
    # Preserve frames if the FPGA resets and its frame counter starts again.
    # Either artifact may remain after an interrupted save or a manual move.
    suffix = 0
    while True:
        stem = base if suffix == 0 else f"{base}_{suffix}"
        path = out / (stem + '.pgm')
        meta_path = out / (stem + '.json')
        if path.exists() or meta_path.exists():
            suffix += 1
            continue
        try:
            handle = path.open('xb') if image_present else meta_path.open('x', encoding='utf-8')
        except FileExistsError:
            suffix += 1
            continue
        break
    meta = {k: v for k, v in frame.items() if k != 'pixels'}
    meta.setdefault('pixel_format', 'gray8')
    meta['image_file'] = path.name if image_present else None
    if image_present:
        with handle:
            handle.write(f"P5\n{frame['width']} {frame['height']}\n255\n".encode('ascii') + frame['pixels'])
        with meta_path.open('x', encoding='utf-8') as handle:
            handle.write(json.dumps(meta, indent=2) + '\n')
    else:
        with handle:
            handle.write(json.dumps(meta, indent=2) + '\n')
    csv_path = _csv_path(out)
    has_header = csv_path.exists() and csv_path.stat().st_size > 0
    with csv_path.open('a', newline='', encoding='utf-8') as handle:
        writer = csv.DictWriter(handle, fieldnames=CSV_FIELDS)
        if not has_header:
            writer.writeheader()
        writer.writerow(_csv_row(meta))
    return path if image_present else meta_path

    preview = PreviewCache() if args.hold_preview else None

                                    f"target_centroid=({target['centroid_x']}, {target['centroid_y']})")
                    print(message)
                    if args.display:
                        picture = (preview.render(frame, cv2, np) if preview is not None
                                   else display_frame(frame, cv2, np))
                        cv2.imshow('FPGA camera', picture)
                        if cv2.waitKey(1) & 0xff == ord('q'):

def packet(frame_id=7, payload=b'FPG1abcd', **changes):
    fields = dict(version=1, mode=1, width=4, height=2, threshold=180,
                  frame_id=frame_id, length=len(payload), bright_count=2,
                  sum_x=7, sum_y=5)
    fields.update(changes)
    h = b'FPG1' + struct.pack('<BBHHHIIIII', *fields.values())
    body = h[4:] + payload
    return h + payload + struct.pack('<H', binascii.crc_hqx(body, 0xffff))


def target_packet(frame_id=70, payload=None, base=None, **changes):

                  frame_id=frame_id, length=24, bright_count=2, sum_x=7, sum_y=5)

// RCLK is free-running at 2 MHz by default, including between frames.
module fifo_capture #(
  parameter integer W=320,H=240, READ_PERIOD=50,
  parameter integer FLUSH_CYCLES=10000,TIMEOUT_CYCLES=100000000,
  parameter integer Y_BYTE=1

  reg[31:0] timer;
  reg[7:0] phase;
  reg[2:0] reset_count;
  reg[18:0] byte_count;
  (* ASYNC_REG="TRUE" *) reg[1:0]vs_sync;
  assign busy=(state!=IDLE);
  always @(posedge clk) begin

          WAIT_BLANK: if(!vs_sync[1])state<=WAIT_FRAME;
          WAIT_FRAME: if(vs_sync[1])begin state<=CAPTURE;cam_wen<=1;end
          CAPTURE: if(!vs_sync[1])begin state<=FLUSH;cam_wen<=0;timer<=0;end

  wire use_features=(OUTPUT_PROTOCOL!=0)?(use_target && features_only):1'b0;
  wire use_packed=(OUTPUT_PROTOCOL!=0)?(pack_edges && edge_mode && !use_features):1'b0;
  wire use_v3=use_features || use_packed;
  wire ready;
  wire valid=(state==HEADER || state==PIXEL || state==CRC_LO || state==CRC_HI);
  wire [7:0] data=(state==HEADER)?header[header_index*8 +: 8]:
    (state==PIXEL)?pixel_byte:(state==CRC_LO)?crc[7:0]:crc[15:8];
  assign busy=(state!=IDLE);
  uart_tx #(.CLKS_PER_BIT(CLKS_PER_BIT)) tx(.clk(clk),.rst(rst),.valid(valid),.data(data),.ready(ready),.tx(uart_txd));
  function [15:0] crc_byte(input [15:0] old_crc,input [7:0] b);
    reg [15:0] c; integer i;
    begin c=old_crc^{b,8'b0};
      for(i=0;i<8;i=i+1) c=c[15]?(c<<1)^16'h1021:c<<1;
      crc_byte=c;
    end
  endfunction
  always @(posedge clk) begin
    if(rst) begin state<=IDLE; done<=0; ram_addr<=0; header<=0; header_index<=0; header_length<=32;
      pixel_index<=0; pixel_byte<=0; packed_output<=0; features_output<=0;
      pack_index<=0; crc<=16'hffff; end
    else begin
      done<=0;
      case(state)
        IDLE: if(start) begin
          header[31:0]<=32'h31475046;
          header[39:32]<=use_v3?8'd3:(use_target?8'd2:8'd1); header[47:40]<={7'b0,edge_mode};
          header[63:48]<=W; header[79:64]<=H;
          header[95:80]<={5'b0,threshold}; header[127:96]<=frame_id;
          header[159:128]<=use_features?0:(use_packed?((PIXELS+7)/8):PIXELS);
          header[191:160]<=bright_count;
          header[223:192]<=sum_x; header[255:224]<=sum_y;
          header[263:256]<={5'b0,config_valid,target_dark,target_valid};
          header[271:264]<=target_threshold;
          header[287:272]<=roi_x_min; header[303:288]<=roi_y_min;
          header[319:304]<=roi_x_max; header[335:320]<=roi_y_max;
          header[351:336]<=bbox_x_min; header[367:352]<=bbox_y_min;
          header[383:368]<=bbox_x_max; header[399:384]<=bbox_y_max;
          header[431:400]<=target_area; header[463:432]<=target_sum_x;
          header[495:464]<=target_sum_y; header[527:496]<=min_area;
          header[559:528]<=max_area;
          // A packed image can omit target detection. Its extension must be
          // deterministic even when target inputs are unconnected or changing.
          if(use_v3 && !use_target) header[559:256]<=0;
          header[567:560]<=use_features?8'd6:(use_packed?(use_target?8'd5:8'd1):8'd0);
          header[575:568]<=0;
          header_length<=(use_v3 || use_target)?7'd72:7'd32;
          packed_output<=use_packed; features_output<=use_features;
          header_index<=0; pixel_index<=0; ram_addr<=0; pixel_byte<=0; pack_index<=0;
          crc<=16'hffff; state<=HEADER;
        end
        HEADER: if(ready) begin
          if(header_index>=4) crc<=crc_byte(crc,data);
          if(header_index==header_length-1'b1) state<=features_output?CRC_LO:RAM_WAIT;
          else header_index<=header_index+1'b1;
        end
        // Address is presented before RAM_WAIT. Capture one full cycle later.
        RAM_WAIT: state<=RAM_CAPTURE;
        RAM_CAPTURE: begin
          if(packed_output) begin
            pixel_byte[pack_index]<=ram_data[7];
            if(pack_index==7 || pixel_index==PIXELS-1) state<=PIXEL;
            else begin
              pack_index<=pack_index+1'b1; pixel_index<=pixel_index+1'b1;
              ram_addr<=ram_addr+1'b1; state<=RAM_WAIT;
            end
          end else begin pixel_byte<=ram_data; state<=PIXEL; end
        end
        PIXEL: if(ready) begin
          crc<=crc_byte(crc,pixel_byte);
          if(pixel_index==PIXELS-1) state<=CRC_LO;
          else begin
            pixel_index<=pixel_index+1'b1; ram_addr<=ram_addr+1'b1;
            pack_index<=0; pixel_byte<=0; state<=RAM_WAIT;
          end
        end
        CRC_LO: if(ready) state<=CRC_HI;
        CRC_HI: if(ready) state<=DRAIN;
        DRAIN: if(ready) begin state<=IDLE; done<=1; end
        default: state<=IDLE;
      endcase
    end
  end
endmodule

        end
        last_output<=absolute_last;
      end
    end
  end
endmodule

    end else begin
      heartbeat<=heartbeat+1'b1;

      if(core_done)core_complete<=1;
      if(target_done)target_complete<=1;
      if(pattern_active)begin
        if(pixel_timer==99)begin

          if(target_enable)
            pattern_pixel<=(px>=W/4 && px<W/2 && py>=H/4 && py<3*H/4)?
              (target_dark?32:220):(target_dark?220:32);
          else pattern_pixel<=(px>=W/4 && px<W/2 && py>=H/4 && py<3*H/4)?220:((px+py)%96);
          if(px==W-1)begin px<=0;if(py==H-1)pattern_active<=0;else py<=py+1'b1;end

        end else pixel_timer<=pixel_timer+1'b1;
      end
      case(state)
        WAITING:if(delay_count<frame_wait_cycles)delay_count<=delay_count+1;
          else if(!sw_sync[15] && (sw_sync[0] || config_done))begin
            pattern_mode<=sw_sync[0];edge_mode<=sw_sync[1];
            threshold<=requested_threshold;
            target_enable<=sw_sync[2];target_dark<=!sw_sync[3];
            // SW10/11/12/13 retain diagnostic-page meanings under SW14.
            preview_active<=preview_requested;
            if(!preview_requested || preview_refresh)preview_count<=0;

            packed_mode<=!sw_sync[14] && sw_sync[10] && sw_sync[1] &&
              !selected_features_only;

            case(sw_sync[9:8])
              2'b00:target_threshold<=96;
              2'b01:target_threshold<=64;

            endcase
            frame_id<=frame_id+1'b1;state<=LAUNCH;
          end
        LAUNCH:begin
          startup_complete<=1;
          core_complete<=0;target_complete<=0;
          if(pattern_mode)begin
            pattern_sof<=1;px<=0;py<=0;pixel_timer<=0;pattern_active<=1;state<=PROCESSING;
          end else begin capture_start<=1;state<=ARM;end
        end
        ARM:if(cap_busy)state<=PROCESSING;
        PROCESSING:if((core_done || core_complete) &&
            (!target_enable || target_done || target_complete))begin tx_start<=1;state<=SENDING;end
          else if(!pattern_mode && cap_error)begin delay_count<=0;state<=WAITING;end
        SENDING:if(tx_done)begin
          if(preview_active)begin
            if(PREVIEW_INTERVAL<=1 || preview_count==PREVIEW_INTERVAL-1)preview_count<=0;
            else preview_count<=preview_count+1'b1;
          end
          delay_count<=0;state<=WAITING;
        end
        default:state<=WAITING;
      endcase
    end
  end
  wire vs_seen,href_seen,wen_seen,overlap_seen,href_vs_high,href_vs_low;
  wire[15:0]captured_lines,high_lines,low_lines;
  camera_monitor monitor_camera(clk100,rst,config_done,cam_vsync,cam_href,cam_wen,
    vs_seen,href_seen,wen_seen,overlap_seen,captured_lines,href_vs_high,href_vs_low,high_lines,low_lines);
  // SW14 selects diagnostics and masks output-speedup switch requests.
  // In diagnostics SW13 chooses high/low line-count byte; normally it enables
  // periodic previews while SW2 and SW11 request target results.
  // Status means observed at FPGA inputs/internal WEN, not at the FIFO chip.
  // SW12 selects register readback page; SW11:10 chooses PID/COM10/TSLB/COM7.
  wire[7:0]register_byte=register_values[sw_sync[11:10]*8+:8];
  wire[15:0]diagnostic_lines=(sw_sync[11:10]==0)?captured_lines:
    (sw_sync[11:10]==1)?high_lines:(sw_sync[11:10]==2)?low_lines:16'b0;
  assign led=(sw_sync[14] && sw_sync[12])?{register_byte,!camera_config_rst,
    camera_fault,cap_error,href_vs_low,href_vs_high,read_error,read_done,config_done}:
    sw_sync[14]?{(sw_sync[13]?diagnostic_lines[15:8]:diagnostic_lines[7:0]),
    !camera_config_rst,camera_fault,cap_error,overlap_seen,wen_seen,href_seen,vs_seen,config_done}:
    {(!sw_sync[0] && (camera_fault||cap_error)),frame_id[7:0],
    heartbeat[26],edge_mode,pattern_mode,tx_busy,cap_busy,config_done,!rst};
endmodule

// Passive diagnostic observer. No connection to acquisition control.
// Kept in this source file so existing Vivado projects need no new RTL source.
module camera_monitor(input wire clk,rst,enabled,vsync,href,wen,
  output reg vs_seen,href_seen,wen_seen,overlap_seen,
  output reg[15:0]captured_lines,output reg href_vs_high,href_vs_low,
  output reg[15:0]high_lines,low_lines);
  (* ASYNC_REG="TRUE" *) reg[1:0]vs_sync=2'b11,href_sync=0;
  reg vs_prev,href_prev,wen_prev;
  reg[15:0]line_count;
  reg[15:0]high_count,low_count;
  always @(posedge clk)begin
    vs_sync<={vs_sync[0],vsync};href_sync<={href_sync[0],href};
    vs_prev<=vs_sync[1];href_prev<=href_sync[1];wen_prev<=wen;
    if(rst || !enabled)begin
      vs_seen<=0;href_seen<=0;wen_seen<=0;overlap_seen<=0;
      captured_lines<=0;line_count<=0;
      href_vs_high<=0;href_vs_low<=0;
      high_count<=0;low_count<=0;high_lines<=0;low_lines<=0;
    end else begin
      if(vs_prev!=vs_sync[1])vs_seen<=1;
      if(href_prev!=href_sync[1])href_seen<=1;
      if(wen)wen_seen<=1;
      if(wen && href_sync[1])overlap_seen<=1;
      // Sample VSYNC at each HREF rising edge, independent of WR.
      if(href_sync[1] && !href_prev)begin
        if(vs_sync[1])href_vs_high<=1;else href_vs_low<=1;
        if(vs_sync[1] && high_count!=16'hffff)high_count<=high_count+1'b1;
        if(!vs_sync[1] && low_count!=16'hffff)low_count<=low_count+1'b1;
      end
      // Latch each completed phase, independent of the capture controller.
      // A simultaneous HREF rise belongs to the newly entered VSYNC phase.
      if(vs_prev && !vs_sync[1])begin high_lines<=high_count;high_count<=0;end
      if(!vs_prev && vs_sync[1])begin low_lines<=low_count;low_count<=0;end
      // A first HREF edge can coincide with WEN becoming active.
      if(wen && !wen_prev)line_count<=(href_sync[1] && !href_prev)?16'd1:16'd0;
      else if(wen && href_sync[1] && !href_prev && line_count!=16'hffff)
        line_count<=line_count+1'b1;
      if(!wen && wen_prev)captured_lines<=line_count;
    end
  end
endmodule

// Single-clock 8N1 transmitter. A byte is accepted only on valid && ready.
module uart_tx #(parameter integer CLKS_PER_BIT=100)(
  input wire clk,rst,valid, input wire [7:0] data,
  output wire ready, output reg tx
);
  localparam integer CW=(CLKS_PER_BIT<2)?1:$clog2(CLKS_PER_BIT);

  reg [9:0] bits;

```python
"""Check release hashes and the generated source listing; --refresh updates both.

Run after editing release files. Regression working directories are intentionally
excluded; copy evidence worth retaining into evidence/ before refreshing.
"""
import argparse
import hashlib
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parents[1]
MANIFEST = 'SHA256.json'
LISTING = '完整代码.md'
SOURCE_DIRS = ('rtl', 'constraints', 'scripts', 'pc', 'sim')
IGNORED_DIRS = {'__pycache__', '.pytest_cache', 'regression', 'xsim.dir'}


def release_files(root):
    """Enumerate deliverables without Vivado scratch files or local test output."""
    files = []
    for folder in (*SOURCE_DIRS, 'docs', 'evidence', 'build'):
        for path in (root / folder).rglob('*'):
            relative = path.relative_to(root)
            if (not path.is_file() or path.suffix in {'.pyc', '.pyo'}
                    or any(part in IGNORED_DIRS for part in relative.parts)):
                continue
            if folder == 'build' and path.suffix not in {'.log', '.rpt', '.bit'}:
                continue
            files.append(path)
    files.extend(p for p in root.iterdir()
                 if p.is_file() and p.suffix in {'.md', '.json'} and p.name != MANIFEST)
    return sorted(files, key=lambda p: p.relative_to(root).as_posix())


def source_listing(root):
    lines = [
        '# 当前Demo完整代码', '',
        '由 `python scripts/check_delivery.py --refresh` 生成，逐文件对应当前源码。',
        '功能、运行方式与验证范围见 `README_先读我.md`；历史硬件记录见 `docs/`。', '',
    ]
    for path in release_files(root):
        relative = path.relative_to(root)
        if relative.parts[0] not in SOURCE_DIRS:
            continue
        if path.suffix not in {'.sv', '.xdc', '.tcl', '.py', '.txt'}:
            continue
        language = {'.sv': 'systemverilog', '.py': 'python', '.tcl': 'tcl',
                    '.xdc': 'tcl'}.get(path.suffix, 'text')
        lines.extend([f'## {relative.as_posix()}', '', f'```{language}',
                      path.read_text(encoding='utf-8').rstrip(), '```', ''])
    return '\n'.join(lines)


def hashes(root):
    return {p.relative_to(root).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in release_files(root)}


def check(root):
    problems = []
    try:
        expected = json.loads((root / MANIFEST).read_text(encoding='utf-8'))
    except (OSError, ValueError) as exc:
        return [f'Cannot read {MANIFEST}: {exc}']
    actual = hashes(root)
    for name in sorted(set(expected) | set(actual)):
        if name not in actual:
            problems.append(f'Missing release file: {name}')
        elif name not in expected:
            problems.append(f'Unlisted release file: {name}')
        elif expected[name] != actual[name]:
            problems.append(f'Hash mismatch: {name}')
    listing = root / LISTING
    if not listing.is_file() or listing.read_text(encoding='utf-8') != source_listing(root):
        problems.append(f'{LISTING} is out of date')
    return problems


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--refresh', action='store_true',
                        help='regenerate source listing and hashes after reviewing changes')
    args = parser.parse_args(argv)
    if args.refresh:
        (ROOT / LISTING).write_text(source_listing(ROOT), encoding='utf-8', newline='\n')
        (ROOT / MANIFEST).write_text(json.dumps(hashes(ROOT), ensure_ascii=False, indent=2)
                                     + '\n', encoding='utf-8', newline='\n')
    problems = check(ROOT)
    if problems:
        print('\n'.join(problems), file=sys.stderr)
        return 1
    print(f'PASS delivery: {len(hashes(ROOT))} file hashes and current source listing')
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
```

```python
"""Run RTL simulations and Python tests without replacing historical evidence.

Tool discovery: explicit --*-bin, environment, PATH, then common install paths.
Use --python-only to run the PC tests on a machine without an RTL simulator.
"""
import argparse
from datetime import datetime
import json
import os
from pathlib import Path
import re
import shutil
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
RTL_TESTS = ('fifo_capture', 'camera_config', 'camera_readback', 'camera_monitor',
             'image_core', 'target_stats', 'transport', 'target_transport',
             'output_transport', 'output_budget', 'top', 'top_target', 'top_output',
             'camera_retry', 'top_recovery', 'top_preview')
WIRE_STREAMS = {
    'target_transport': ('target_wire.bin', 6),
    'top_target': ('top_target_wire.bin', 4),
    'output_transport': ('output_wire.bin', 6),
    'top_output': ('top_output_wire.bin', 5),
    'top_recovery': ('top_recovery_wire.bin', 3),
    'top_preview': ('top_preview_wire.bin', 13),
}
ERROR_LINE = re.compile(r'^\s*(?:fatal|error)\s*(?::|\[)', re.I | re.M)
PASS_LINE = re.compile(r'^PASS(?:\b|_)', re.M)


class RegressionError(RuntimeError):
    """An actionable regression failure, shown without a Python traceback."""


def tool_names(name):
    suffixes = ('.bat', '.exe', '') if os.name == 'nt' else ('',)
    return tuple(name + suffix for suffix in suffixes)


def tools_in(directory, names):
    """Accept either an install root or its bin directory, as a complete set."""
    directory = Path(directory).expanduser()
    for candidate in (directory, directory / 'bin'):
        found = {}
        for name in names:
            path = next((candidate / filename for filename in tool_names(name)
                         if (candidate / filename).is_file()), None)
            if path is None:
                break
            found[name] = path.resolve()
        if len(found) == len(names):
            return found
    return None


def common_install_dirs(simulator):
    """Only inspect known install layouts; do not recursively scan user disks."""
    if os.name != 'nt':
        if simulator == 'vivado':
            for base in (Path('/opt/Xilinx/Vivado'), Path('/tools/Xilinx/Vivado')):
                yield from sorted(base.glob('*/bin'), reverse=True)
        return
    for drive in ('C:', 'D:', 'E:'):
        if simulator == 'vivado':
            for base in ('Xilinx/Vivado', 'AMD/Vivado', 'Vivado'):
                root = Path(drive + '/') / base
                yield from sorted(root.glob('*/bin'), reverse=True)
                yield from sorted(root.glob('*/Vivado/bin'), reverse=True)
            root = Path(drive + '/') / 'AMDDesignTools'
            yield from sorted(root.glob('*/Vivado/bin'), reverse=True)
        else:
            for base in ('iverilog/bin', 'Icarus Verilog/bin',
                         'Program Files/iverilog/bin'):
                yield Path(drive + '/') / base


def discover_tools(explicit, env_names, names, simulator):
    if explicit is not None:
        result = tools_in(explicit, names)
        if result is None:
            raise RegressionError(
                f'{simulator}: {explicit} must contain {", ".join(names)} '
                '(or a bin directory containing them).')
        return result
    for env_name in env_names:
        value = os.environ.get(env_name)
        if value:
            result = tools_in(value, names)
            if result:
                return result
    found = {}
    for name in names:
        path = next((p for filename in tool_names(name)
                     if (p := shutil.which(filename))), None)
        if path is None:
            break
        found[name] = Path(path).resolve()
    if len(found) == len(names):
        return found
    for directory in common_install_dirs(simulator):
        result = tools_in(directory, names)
        if result:
            return result
    return None


def discover_simulator(args):
    selected = args.simulator
    if args.vivado_bin is not None:
        selected = 'vivado'
    elif args.iverilog_bin is not None:
        selected = 'iverilog'
    if selected in ('auto', 'vivado'):
        found = discover_tools(args.vivado_bin,
                               ('VIVADO_BIN', 'XILINX_VIVADO', 'VIVADO_HOME'),
                               ('xvlog', 'xelab', 'xsim'), 'vivado')
        if found:
            return 'vivado', found
    if selected in ('auto', 'iverilog'):
        found = discover_tools(args.iverilog_bin, ('IVERILOG_BIN',),
                               ('iverilog', 'vvp'), 'iverilog')
        if found:
            return 'iverilog', found
    raise RegressionError(
        f'No complete {selected} RTL simulator found. Use --vivado-bin BIN '
        '(or XILINX_VIVADO/VIVADO_BIN), --iverilog-bin BIN '
        '(or IVERILOG_BIN), or put the tools on PATH. '
        'Use --python-only to run the PC tests without RTL simulation.')


def run(command, cwd, log, timeout):
    command = [str(c) for c in command]
    try:
        result = subprocess.run(command, cwd=cwd, stdout=subprocess.PIPE,
                                stderr=subprocess.STDOUT, encoding='utf-8',
                                errors='replace', timeout=timeout)
    except subprocess.TimeoutExpired as exc:
        output = exc.stdout or ''
        if isinstance(output, bytes):
            output = output.decode('utf-8', errors='replace')
        log.write_text(output + f'\nTIMEOUT after {timeout:g} seconds\n',
                       encoding='utf-8')
        raise RegressionError(f'Timeout after {timeout:g} seconds; see {log}') from exc
    except OSError as exc:
        log.write_text(str(exc) + '\n', encoding='utf-8')
        raise RegressionError(f'Could not start {command[0]}: {exc}; see {log}') from exc
    log.write_text(result.stdout, encoding='utf-8')
    # Some simulators report $fatal/$error but return exit status zero.
    if result.returncode or ERROR_LINE.search(result.stdout):
        # Keep the full UTF-8 log even when a legacy Windows console cannot
        # print a tool's Unicode diagnostics. Reporting must not mask failure.
        printable = result.stdout
        encoding = getattr(sys.stdout, 'encoding', None)
        if encoding:
            printable = printable.encode(encoding, errors='backslashreplace').decode(encoding)
        print(printable, end='' if printable.endswith('\n') else '\n')
        raise RegressionError(f'Command failed (exit {result.returncode}); see {log}')
    return result.stdout


def run_python_tests(output, timeout):
    for directory, pattern, logfile in (
            ('pc', 'test_*.py', 'python_tests.log'),
            ('scripts', 'test_*.py', 'runner_tests.log')):
        run([sys.executable, '-B', '-m', 'unittest', 'discover',
             '-s', ROOT / directory, '-p', pattern, '-v'],
            ROOT, output / logfile, timeout)
    print('PASS Python receiver and regression-runner unittest suites', flush=True)


def run_rtl(name, simulator, tools, output, timeout):
    dest = output / ('verify_' + name)
    dest.mkdir()
    sources = sorted((ROOT / 'rtl').glob('*.sv')) + [ROOT / 'sim' / f'tb_{name}.sv']
    if simulator == 'vivado':
        run([tools['xvlog'], '-sv', *sources], dest, dest / 'compile.log', timeout)
        run([tools['xelab'], f'tb_{name}', '-s', 'test'],
            dest, dest / 'elaborate.log', timeout)
        command = [tools['xsim'], 'test', '-runall']
    else:
        image = dest / 'test.vvp'
        run([tools['iverilog'], '-g2012', '-s', f'tb_{name}',
             '-o', image, *sources], dest, dest / 'compile.log', timeout)
        command = [tools['vvp'], image]
    result = run(command, dest, dest / 'simulate.log', timeout)
    if not PASS_LINE.search(result):
        raise RegressionError(f'Missing PASS marker; see {dest / "simulate.log"}')
    print(next(line for line in result.splitlines() if PASS_LINE.match(line)), flush=True)
    return dest


def existing_wire():
    candidates = sorted((ROOT / 'build' / 'regression').glob('*/verify_transport/wire.bin'),
                        reverse=True)
    candidates.append(ROOT / 'build' / 'verify_transport' / 'wire.bin')

def arguments(argv):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--python-only', action='store_true',
                        help='run Python suites and any available wire cross-check; skip RTL')
    parser.add_argument('--simulator', choices=('auto', 'vivado', 'iverilog'), default='auto')
    parser.add_argument('--vivado-bin', type=Path, help='Vivado installation root or bin directory')
    parser.add_argument('--iverilog-bin', type=Path, help='Icarus installation root or bin directory')
    parser.add_argument('--output-dir', type=Path,
                        help='new output directory (default: build/regression/TIMESTAMP)')
    parser.add_argument('--wire', type=Path,
                        help='existing wire stream to check (default: this run or latest saved wire.bin)')

    parser.add_argument('--tests', nargs='+', choices=RTL_TESTS, default=RTL_TESTS,

    parser.add_argument('--timeout', type=float, default=180,
                        help='per-command timeout in seconds (default: 180)')
    args = parser.parse_args(argv)
    if args.timeout <= 0 or args.wire_count <= 0:
        parser.error('--timeout and --wire-count must be positive')
    if args.vivado_bin is not None and args.iverilog_bin is not None:
        parser.error('choose only one of --vivado-bin and --iverilog-bin')
    if args.vivado_bin is not None and args.simulator == 'iverilog':

    if args.iverilog_bin is not None and args.simulator == 'vivado':

def main(argv=None):
    args = arguments(argv)
    if args.wire is not None and not args.wire.is_file():
        print(f'FAIL: Requested wire stream does not exist: {args.wire}', file=sys.stderr)
        return 1
    output = (args.output_dir or ROOT / 'build' / 'regression' /
              datetime.now().strftime('%Y%m%d_%H%M%S_%f')).resolve()
    try:
        output.mkdir(parents=True, exist_ok=False)
    except OSError as exc:
        print(f'FAIL: Need a new writable output directory: {output}: {exc}', file=sys.stderr)
        return 1
    print(f'Output: {output}', flush=True)
    summary = {'status': 'running', 'output': str(output),
               'python': {'status': 'running'}, 'rtl': {'status': 'pending'},
               'wire': {'status': 'pending'}, 'feature_wires': []}
    wire = args.wire.resolve() if args.wire is not None else None
    try:
        run_python_tests(output, args.timeout)
        summary['python'] = {'status': 'passed'}
        if args.python_only:
            summary['rtl'] = {'status': 'skipped', 'reason': '--python-only'}
            print('SKIP RTL simulation (--python-only)', flush=True)
        else:
            summary['rtl'] = {'status': 'running'}
            simulator, tools = discover_simulator(args)
            summary['rtl'] = {'status': 'running', 'simulator': simulator,
                              'tools': {k: str(v) for k, v in tools.items()}, 'tests': []}
            print(f'RTL simulator: {simulator}', flush=True)
            for name in dict.fromkeys(args.tests):
                dest = run_rtl(name, simulator, tools, output, args.timeout)
                summary['rtl']['tests'].append(name)
                if name == 'transport' and wire is None:
                    wire = dest / 'wire.bin'
                    if not wire.is_file():
                        raise RegressionError(f'tb_transport did not create {wire}')
                if name in WIRE_STREAMS:
                    filename, count = WIRE_STREAMS[name]
                    feature_wire = dest / filename
                    if not feature_wire.is_file():
                        raise RegressionError(f'tb_{name} did not create {feature_wire}')
                    summary['feature_wires'].append({
                        'status': 'pending', 'name': name, 'input': str(feature_wire),
                        'required_frames': count})
            summary['rtl']['status'] = 'passed'
        wire = wire or existing_wire()
        if wire is None:
            summary['wire'] = {'status': 'skipped', 'reason': 'No recorded wire.bin is available'}
            print('SKIP wire cross-check: no wire.bin available; use --wire PATH or run transport RTL.',
                  flush=True)
        else:
            summary['wire'] = {'status': 'running', 'input': str(wire),
                               'required_frames': args.wire_count}
            run([sys.executable, '-B', ROOT / 'pc' / 'receiver.py', '--input', wire,
                 '--out', output / 'verified_frames', '--count', args.wire_count],
                ROOT, output / 'wire_crosscheck.log', args.timeout)
            summary['wire']['status'] = 'passed'
            print(f'PASS receiver wire-stream cross-check ({args.wire_count} frames): {wire}',
                  flush=True)
        for feature in summary['feature_wires']:
            feature['status'] = 'running'
            run([sys.executable, '-B', ROOT / 'pc' / 'receiver.py', '--input', feature['input'],
                 '--out', output / ('verified_' + feature['name']),
                 '--count', feature['required_frames']],
                ROOT, output / (feature['name'] + '_crosscheck.log'), args.timeout)
            feature['status'] = 'passed'
            print(f'PASS protocol wire-stream cross-check: {feature["name"]}', flush=True)
        summary['status'] = 'passed'
        print('PASS requested regression checks; see summary.json for passes and skips.', flush=True)
        return 0
    except (RegressionError, OSError) as exc:
        summary['status'] = 'failed'
        summary['error'] = str(exc)
        for section in ('python', 'rtl', 'wire'):
            if summary[section]['status'] == 'running':
                summary[section]['status'] = 'failed'
        for feature in summary['feature_wires']:
            if feature['status'] == 'running':
                feature['status'] = 'failed'
        print(f'FAIL: {exc}', file=sys.stderr)
        return 1
    finally:
        (output / 'summary.json').write_text(
            json.dumps(summary, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')

module tb_fifo_capture;
  reg clk=0; always #5 clk=~clk;
  reg rst=1,start=0,vsync=1;
  wire rclk,rrst_n,wen,valid,sof,done,busy,error;
  reg [7:0] data=0;
  wire [7:0] pixel;
  integer ptr=0,got=0,frames=0,sofs=0,completions=0,rclk_edges=0;
  fifo_capture #(.W(4),.H(3),.READ_PERIOD(50),.FLUSH_CYCLES(100),.TIMEOUT_CYCLES(1000))
    dut(clk,rst,start,vsync,data,rclk,rrst_n,wen,sof,valid,pixel,done,busy,error);
  // Behavioral AL422 model: reset is synchronous to read clock;
  // first enabled rising edge after reset presents byte zero.
  always @(posedge rclk) begin
    rclk_edges=rclk_edges+1;
    if(!rrst_n) begin ptr=0; data<=0; end
    else begin data <= (ptr%2) ? (ptr/2+frames*20) : 8'hee; ptr=ptr+1; end
  end
  always @(posedge clk) if(!rst) begin
    if(sof)sofs=sofs+1;
    if(done)completions=completions+1;
    if(valid) begin
      if(pixel !== (got+frames*20)) $fatal(1,"byte/position error got %0d value %0d",got,pixel);
      got=got+1;
    end
  end
  task launch;
    begin
      @(negedge clk);start=1;@(negedge clk);start=0;
      if(!busy || error)$fatal(1,"new capture must accept start and clear previous timeout");
    end
  endtask
  task supply_frame;
    begin
      repeat(10) @(negedge clk);vsync=0;
      repeat(10) @(negedge clk);vsync=1;
      repeat(10) @(negedge clk);
      if(!wen) $fatal(1,"capture not armed");
      repeat(100) @(negedge clk);vsync=0;
    end
  endtask
  task capture;
    integer before_sofs,before_completions;
    begin
      before_sofs=sofs;before_completions=completions;
      launch();supply_frame();
      wait(done); repeat(2) @(negedge clk);
      if(got!=12 || error || wen || busy || !rrst_n ||
          sofs!=before_sofs+1 || completions!=before_completions+1)
        $fatal(1,"frame not complete after recovery");
      vsync=1; got=0; frames=frames+1;
      repeat(20) @(negedge clk);
    end
  endtask
  task timeout_in_phase(input integer stuck_phase);
    integer before_sofs,before_completions,before_edges;
    begin
      before_sofs=sofs;before_completions=completions;
      vsync=1;launch();
      if(stuck_phase>=2)begin
        repeat(10)@(negedge clk);vsync=0;
        repeat(10)@(negedge clk);
      end
      if(stuck_phase==3)begin
        vsync=1;repeat(10)@(negedge clk);
        if(!wen)$fatal(1,"CAPTURE timeout case never enabled writes");
      end
      if(dut.state!=stuck_phase)$fatal(1,"wrong timeout phase %0d",dut.state);
      wait(error);@(negedge clk);
      if(wen || busy || !rrst_n || sof || valid || done || got!=0)
        $fatal(1,"phase %0d timeout must release capture without output",stuck_phase);
      before_edges=rclk_edges;
      repeat(150)@(negedge clk);
      if(!error || wen || busy || !rrst_n || sofs!=before_sofs ||
          completions!=before_completions || got!=0 || rclk_edges-before_edges<2)
        $fatal(1,"phase %0d timeout must remain idle with free-running RCLK",stuck_phase);
      vsync=1;repeat(10)@(negedge clk);
      capture();
      $display("PASS fifo_capture phase %0d timeout, pin release and next-frame recovery",stuck_phase);
    end
  endtask
  integer before_reset_completions;
  initial begin
    repeat(5) @(negedge clk); rst=0;
    capture(); capture();
    timeout_in_phase(1); // VSYNC stuck high before blanking.
    timeout_in_phase(2); // VSYNC stuck low before active frame.
    timeout_in_phase(3); // VSYNC remains high while writes are enabled.
    before_reset_completions=completions;
    launch();supply_frame();wait(got==5);
    @(negedge clk);rst=1;
    repeat(3)@(negedge clk);
    if(busy || wen || !rrst_n || rclk || sof || valid || done || error ||
        completions!=before_reset_completions)
      $fatal(1,"reset during FIFO read must discard partial frame and release pins");
    got=0;vsync=1;rst=0;
    repeat(10)@(negedge clk);capture();
    if(frames!=6 || completions!=6 || sofs!=7)
      $fatal(1,"unexpected completed/partial frame totals %0d/%0d/%0d",frames,completions,sofs);
    $display("PASS fifo_capture six complete frames, three timeout phases and mid-read reset recovery");
    $finish;
  end
  initial begin #1000000; $fatal(1,"test timeout");end
endmodule

module output_budget_case #(
  parameter integer W=318,H=238,CPB=4
)(
  output reg complete=0,
  output reg [31:0] raw_bytes=0,packed_bytes=0,features_bytes=0,
  output reg [31:0] raw_cycles=0,packed_cycles=0,features_cycles=0
);
  localparam integer N=W*H;
  reg clk=0,rst=1,start=0,pack_edges=0,features_only=0;
  always #5 clk=~clk;
  reg [31:0] frame_id=900;
  wire [16:0] addr;
  reg [7:0] rd;
  wire busy,done,tx;
  integer scenario=0,accepted=0,captures=0,cycles=0,expected_bytes=0;
  reg [15:0] crc=16'hffff;
  frame_sender #(.W(W),.H(H),.CLKS_PER_BIT(CPB),.TARGET_PROTOCOL(1),.OUTPUT_PROTOCOL(1)) dut
    (.clk(clk),.rst(rst),.start(start),.edge_mode(1'b1),.threshold(11'd180),
     .frame_id(frame_id),.bright_count(32'd0),.sum_x(32'd0),.sum_y(32'd0),
     .ram_addr(addr),.ram_data(rd),.busy(busy),.done(done),.uart_txd(tx),
     .target_enable(1'b1),.target_valid(1'b0),.target_dark(1'b1),.config_valid(1'b1),
     .target_threshold(8'd64),.roi_x_min(16'd1),.roi_y_min(16'd1),
     .roi_x_max(16'(W)),.roi_y_max(16'(H)),.bbox_x_min(16'd0),.bbox_y_min(16'd0),
     .bbox_x_max(16'd0),.bbox_y_max(16'd0),.target_area(32'd0),.target_sum_x(32'd0),
     .target_sum_y(32'd0),.min_area(32'd1),.max_area(32'(N)),
     .pack_edges(pack_edges),.features_only(features_only));
  function [7:0] memory_byte(input integer index);
    begin memory_byte=(index%3==0 || index==N-1)?8'hff:8'h00; end
  endfunction
  function [15:0] update_crc(input [15:0] c,input [7:0] b);
    reg [15:0] v; integer k;
    begin v=c^{b,8'b0}; for(k=0;k<8;k=k+1)v=v[15]?(v<<1)^16'h1021:v<<1;update_crc=v;end
  endfunction
  function [7:0] expected_byte(input integer index);
    reg [7:0] b;
    integer k,payload_length;
    reg [31:0] val;
    begin
      b=0;val=0;
      payload_length=(scenario==2)?0:((scenario==1)?((N+7)/8):N);
      case(index)
        0:b="F";1:b="P";2:b="G";3:b="1";
        4:b=(scenario==0)?2:3;5:b=1;
        6:b=W&255;7:b=(W>>8)&255;8:b=H&255;9:b=(H>>8)&255;
        10:b=180;32:b=6;33:b=64;
        34:b=1;36:b=1;38:b=W&255;39:b=(W>>8)&255;
        40:b=H&255;41:b=(H>>8)&255;62:b=1;
        70:b=(scenario==0)?0:((scenario==1)?5:6);
        default:begin end
      endcase
      if(index>=12 && index<=15)begin val=900+scenario;b=(val>>(8*(index-12)))&255;end
      if(index>=16 && index<=19)begin val=payload_length;b=(val>>(8*(index-16)))&255;end
      if(index>=66 && index<=69)begin val=N;b=(val>>(8*(index-66)))&255;end
      if(index>=72)begin
        if(scenario==0)b=memory_byte(index-72);
        else begin
          b=0;
          for(k=0;k<8;k=k+1)
            if((index-72)*8+k<N)b[k]=memory_byte((index-72)*8+k)==255;
        end
      end
      expected_byte=b;
    end
  endfunction
  always @(posedge clk)begin
    rd<=memory_byte(addr);

    else begin
      if(busy)cycles=cycles+1;
      if(dut.state==3)begin
        if(addr!==captures)$fatal(1,"budget N=%0d scenario=%0d RAM address %0d expected %0d",N,scenario,addr,captures);
        captures=captures+1;

      if(dut.valid && dut.ready)begin
        if(accepted<expected_bytes-2)begin
          if(dut.data!==expected_byte(accepted))
            $fatal(1,"budget N=%0d scenario=%0d byte=%0d got=%h expected=%h",N,scenario,accepted,dut.data,expected_byte(accepted));
          if(accepted>=4)crc=update_crc(crc,dut.data);
        end else if(accepted==expected_bytes-2)begin
          if(dut.data!==crc[7:0])$fatal(1,"budget CRC low N=%0d",N);
        end else if(accepted==expected_bytes-1)begin
          if(dut.data!==crc[15:8])$fatal(1,"budget CRC high N=%0d",N);
        end else $fatal(1,"budget excess byte N=%0d",N);
        accepted=accepted+1;
      end
    end
  end
  initial begin
    repeat(5)@(negedge clk);rst=0;
    for(scenario=0;scenario<3;scenario=scenario+1)begin
      pack_edges=(scenario!=0);features_only=(scenario==2);frame_id=900+scenario;
      expected_bytes=74+((scenario==0)?N:((scenario==1)?((N+7)/8):0));
      @(negedge clk);start=1;@(negedge clk);start=0;
      // The bench verifies the complete packet even when requests change live.
      pack_edges=~pack_edges;features_only=~features_only;frame_id=0;
      wait(done);@(negedge clk);
      if(accepted!==expected_bytes)$fatal(1,"budget byte count N=%0d",N);
      if(captures!==((scenario==2)?0:N))$fatal(1,"budget RAM count N=%0d",N);
      if(busy)$fatal(1,"budget busy after done");
      case(scenario)
        0:begin raw_bytes=accepted;raw_cycles=cycles;end
        1:begin packed_bytes=accepted;packed_cycles=cycles;end
        2:begin features_bytes=accepted;features_cycles=cycles;end
      endcase
    end
    if(N>8 && !(features_cycles<packed_cycles && packed_cycles<raw_cycles))
      $fatal(1,"budget cycles did not improve N=%0d",N);
    complete=1;
  end
endmodule

        put_u32(62,min_area);put_u32(66,max_area);

      end
      crc=16'hffff;

      expected[total-2]=crc[7:0];expected[total-1]=crc[15:8];
    end
  endtask

    begin
      mode=~mode;pack_edges=~pack_edges;features_only=~features_only;
      threshold=0;frame_id=32'hdeadbeef;bc=0;sx=0;sy=0;
      target_enable=~target_enable;target_valid=~target_valid;

      min_area=~min_area;max_area=~max_area;
    end

    rst=1;repeat(3)@(negedge clk);
    if(busy || tx!==1 || done)$fatal(1,"reset did not clear active packet");
    rst=0;
    for(f=0;f<6;f=f+1)begin
      configure_frame(f);make_expected();
      @(negedge clk);start=1;
      @(negedge clk);start=0;mutate_live_inputs();
      for(i=0;i<total;i=i+1)begin get_byte(received[i]);$fwrite(fd,"%c",received[i]);end
      for(i=0;i<total;i=i+1)
        if(received[i]!==expected[i])
          $fatal(1,"output frame %0d byte %0d got %h expected %h",f,i,received[i],expected[i]);
      wait(done);@(negedge clk);
      if(busy)$fatal(1,"busy after done");
      if(ram_captures!==(expected_features?0:PIXELS))$fatal(1,"RAM count frame %0d: %0d",f,ram_captures);
    end
    $fclose(fd);
    $display("PASS output transport six frames: packed LSB/tail, target/no target, features priority, fallbacks, reset, latching, RAM sequence, UART and CRC");
    $finish;
  end
  initial begin #1000000;$fatal(1,"output transport timeout");end

  task put_u16(input integer offset,input [15:0] value);
    begin expected[offset]=value[7:0];expected[offset+1]=value[15:8];end
  endtask
  task put_u32(input integer offset,input [31:0] value);
    integer k;
    begin for(k=0;k<4;k=k+1)expected[offset+k]=(value>>(8*k))&8'hff;end
  endtask
  task get_byte(output [7:0] b);
    integer k;
    begin
      @(negedge tx); #(CPB*10/2);
      if(tx!==0) $fatal(1,"bad start bit");
      for(k=0;k<8;k=k+1) begin #(CPB*10); b[k]=tx; end
      #(CPB*10); if(tx!==1) $fatal(1,"bad stop bit");
    end
  endtask
  task configure_frame(input integer frame);
    begin
      mode=frame[0];threshold=11'h5ab+frame;frame_id=32'h10203040+frame;
      bc=4;sx=10;sy=8;
      target_enable=1;target_valid=0;target_dark=1;config_valid=1;
      target_threshold=64;roi_x_min=1;roi_y_min=1;roi_x_max=W;roi_y_max=H;
      bbox_x_min=0;bbox_y_min=0;bbox_x_max=0;bbox_y_max=0;
      target_area=0;target_sum_x=0;target_sum_y=0;min_area=1;max_area=6;
      case(frame)
        0:begin // Valid dark target, in original sensor coordinates.
          target_valid=1;target_area=2;target_sum_x=3;target_sum_y=2;
          bbox_x_min=1;bbox_y_min=1;bbox_x_max=2;bbox_y_max=1;
        end
        1:begin // Valid bright configuration, with no matching target.
          target_dark=0;target_threshold=250;roi_x_min=2;
        end
        2:begin // Invalid configuration retains raw settings, zero statistics.
          config_valid=0;target_threshold=173;
          roi_x_min=16'h0102;roi_y_min=16'h0203;
          min_area=32'h00030004;max_area=32'h00020003;
        end
        3:begin // Geometry remains available when minimum-area filtering fails.
          target_threshold=96;target_area=1;target_sum_x=2;target_sum_y=1;
          bbox_x_min=2;bbox_y_min=1;bbox_x_max=2;bbox_y_max=1;min_area=2;
        end
        4:begin // Return to legacy format after four extended headers.
          target_enable=0;target_valid=1;target_area=2;target_sum_x=3;target_sum_y=2;
          bbox_x_min=1;bbox_y_min=1;bbox_x_max=2;bbox_y_max=1;
        end
        5:begin // Re-enable v2, preserving statistics above the maximum area.
          target_dark=0;target_threshold=128;target_area=8;target_sum_x=20;target_sum_y=12;
          bbox_x_min=1;bbox_y_min=1;bbox_x_max=W;bbox_y_max=H;
        end
      endcase
    end
  endtask
  task make_expected;
    integer k;
    begin
      for(k=0;k<72;k=k+1)expected[k]=0;
      expected[0]="F";expected[1]="P";expected[2]="G";expected[3]="1";
      expected[4]=target_enable?2:1;expected[5]=mode;
      put_u16(6,W);put_u16(8,H);put_u16(10,{5'b0,threshold});
      put_u32(12,frame_id);put_u32(16,W*H);put_u32(20,bc);put_u32(24,sx);put_u32(28,sy);
      expected[32]={5'b0,config_valid,target_dark,target_valid};expected[33]=target_threshold;
      put_u16(34,roi_x_min);put_u16(36,roi_y_min);put_u16(38,roi_x_max);put_u16(40,roi_y_max);
      put_u16(42,bbox_x_min);put_u16(44,bbox_y_min);put_u16(46,bbox_x_max);put_u16(48,bbox_y_max);
      put_u32(50,target_area);put_u32(54,target_sum_x);put_u32(58,target_sum_y);
      put_u32(62,min_area);put_u32(66,max_area);put_u16(70,0);
    end
  endtask
  task mutate_live_inputs;
    begin
      mode=~mode;threshold=0;frame_id=32'hdeadbeef;bc=0;sx=0;sy=0;
      target_enable=~target_enable;target_valid=~target_valid;
      target_dark=~target_dark;config_valid=~config_valid;target_threshold=~target_threshold;
      roi_x_min=~roi_x_min;roi_y_min=~roi_y_min;roi_x_max=~roi_x_max;roi_y_max=~roi_y_max;
      bbox_x_min=~bbox_x_min;bbox_y_min=~bbox_y_min;bbox_x_max=~bbox_x_max;bbox_y_max=~bbox_y_max;
      target_area=~target_area;target_sum_x=~target_sum_x;target_sum_y=~target_sum_y;
      min_area=~min_area;max_area=~max_area;
    end
  endtask
  initial begin
    fd=$fopen("target_wire.bin","wb");
    if(fd==0)$fatal(1,"cannot open target_wire.bin");
    for(i=0;i<W*H;i=i+1)mem[i]=i*31+5;
    repeat(5)@(negedge clk);rst=0;
    for(f=0;f<6;f=f+1)begin
      configure_frame(f);make_expected();
      header_bytes=(f==4)?32:72;total=header_bytes+W*H+2;
      @(negedge clk);start=1;
      @(negedge clk);start=0;mutate_live_inputs();
      for(i=0;i<total;i=i+1)begin get_byte(received[i]);$fwrite(fd,"%c",received[i]);end
      for(i=0;i<header_bytes;i=i+1)
        if(received[i]!==expected[i])
          $fatal(1,"target header frame %0d byte %0d got %h expected %h",f,i,received[i],expected[i]);
      for(i=0;i<W*H;i=i+1)
        if(received[header_bytes+i]!==mem[i])$fatal(1,"target frame %0d pixel %0d",f,i);
      crc=16'hffff;
      for(i=4;i<total-2;i=i+1)crc=update_crc(crc,received[i]);
      if({received[total-1],received[total-2]}!==crc)$fatal(1,"target frame %0d CRC",f);
      wait(done);@(negedge clk);
      if(busy)$fatal(1,"target busy after done");
    end
    $fclose(fd);
    $display("PASS target transport six frames: valid, absent, invalid config, area filters, v1/v2 transition; all metadata latched");
    $finish;
  end
  initial begin #1000000;$fatal(1,"target transport timeout");end
endmodule

`timescale 1ns/1ps
// Complete top-level UART path, including the switches, output framebuffer,
// first-frame delay, and every runtime transition back to an image.
module tb_top_output #(parameter integer FAST_CYCLES=3);
  localparam integer W=9,H=6,OW=W-2,OH=H-2,N=OW*OH,CPB=4,MIN_AREA=4;
  localparam integer START_CYCLES=17,FRAME_CYCLES=29;
  reg clk=0;always #5 clk=~clk;
  reg btnC=1;reg[15:0]sw=16'h0001;
  tri1 scl,sda,cam_reset;
  wire tx,rclk,rrst,wen;wire[15:0]led;
  stream_top #(.W(W),.H(H),.UART_DIV(CPB),.START_DELAY(START_CYCLES),
    .FRAME_DELAY(FRAME_CYCLES),.FAST_FRAME_DELAY(FAST_CYCLES),.TARGET_MIN_AREA(MIN_AREA),
    .CAM_RESET_CYCLES(10),.CAM_SETTLE_CYCLES(12)) dut(
    .clk100(clk),.btnC(btnC),.sw(sw),.cam_data(8'b0),.cam_vsync(1'b1),.cam_href(1'b0),
    .cam_rst_n(cam_reset),.cam_sioc(scl),.cam_siod(sda),.cam_rclk(rclk),
    .cam_rrst_n(rrst),.cam_wen(wen),.uart_txd(tx),.led(led));

  reg[7:0]received[0:72+N+2-1];
  reg[7:0]frozen_framebuffer[0:N-1];
  reg[7:0]frozen_read_data;
  reg watching_features=0;
  integer writes[0:4],core_pixels[0:4];
  integer fd,f,i,j,k,header_bytes,payload_bytes,total,expected_byte,xx,yy;
  integer control_frame,cycle=0,wait_enter_cycle=0,launch_count=0,expected_delay;
  integer active_frame,reference_pixel,snapshot_index;
  integer ref_bright,ref_bright_x,ref_bright_y,ref_area,ref_x,ref_y;
  integer ref_x_min,ref_y_min,ref_x_max,ref_y_max;
  reg previous_rst=1,previous_fast=0;
  reg[2:0]previous_state=0;
  reg[15:0]crc;

  function[15:0] update_crc(input[15:0]c,input[7:0]b);
    reg[15:0]v;integer n;
    begin v=c^{b,8'b0};for(n=0;n<8;n=n+1)v=v[15]?(v<<1)^16'h1021:v<<1;update_crc=v;end
  endfunction
  function[31:0]u32(input integer offset);
    u32={received[offset+3],received[offset+2],received[offset+1],received[offset]};
  endfunction
  function[15:0]u16(input integer offset);
    u16={received[offset+1],received[offset]};
  endfunction
  // Independent, coordinate-based model of the synthetic source and Sobel.
  function integer source_y(input integer frame,x,y);
    reg inside_rect;
    begin
      inside_rect=(x>=W/4 && x<W/2 && y>=H/4 && y<3*H/4);
      if(frame==0)source_y=inside_rect?220:((x+y)%96);
      else if(frame==2 || frame==3)source_y=inside_rect?220:32;
      else source_y=inside_rect?32:220;
    end
  endfunction
  function integer image_y(input integer frame,x,y);
    integer gx,gy,mag;
    begin
      if(frame==0 || frame==3)image_y=source_y(frame,x,y);
      else begin
        gx=source_y(frame,x+1,y-1)+2*source_y(frame,x+1,y)+source_y(frame,x+1,y+1)
          -source_y(frame,x-1,y-1)-2*source_y(frame,x-1,y)-source_y(frame,x-1,y+1);
        gy=source_y(frame,x-1,y+1)+2*source_y(frame,x,y+1)+source_y(frame,x+1,y+1)
          -source_y(frame,x-1,y-1)-2*source_y(frame,x,y-1)-source_y(frame,x+1,y-1);
        mag=(gx<0?-gx:gx)+(gy<0?-gy:gy);
        image_y=(mag>=((frame==2)?960:160))?255:0;
      end
    end
  endfunction
  task build_reference(input integer frame);
    integer x,y,value;reg selected;
    begin
      ref_bright=0;ref_bright_x=0;ref_bright_y=0;
      ref_area=0;ref_x=0;ref_y=0;
      ref_x_min=0;ref_y_min=0;ref_x_max=0;ref_y_max=0;
      for(y=1;y<=OH;y=y+1)for(x=1;x<=OW;x=x+1)begin
        value=source_y(frame,x,y);
        if(value>=128)begin
          ref_bright=ref_bright+1;ref_bright_x=ref_bright_x+x;ref_bright_y=ref_bright_y+y;
        end
        selected=(frame==2 || frame==3)?(value>=96):(value<=96);
        if(frame!=0 && selected)begin
          if(ref_area==0)begin ref_x_min=x;ref_x_max=x;ref_y_min=y;ref_y_max=y;end
          else begin
            if(x<ref_x_min)ref_x_min=x;if(x>ref_x_max)ref_x_max=x;
            if(y<ref_y_min)ref_y_min=y;if(y>ref_y_max)ref_y_max=y;
          end
          ref_area=ref_area+1;ref_x=ref_x+x;ref_y=ref_y+y;
        end
      end
    end
  endtask
  task check_controls(input integer frame);
    begin
      if(dut.pattern_mode!==1'b1 || dut.edge_mode!==(frame==1 || frame==2 || frame==4) ||
         dut.target_enable!==(frame!=0) || dut.target_dark!==!(frame==2 || frame==3) ||
         dut.threshold!==((frame==2)?11'd960:11'd160) || dut.target_threshold!==8'd96 ||
         dut.packed_mode!==(frame==1) || dut.features_only!==(frame==2) ||
         dut.fast_frames!==(frame>=1 && frame<=3))
        $fatal(1,"top output controls changed inside frame %0d",frame);
    end
  endtask
  task get_byte(output[7:0]b);
    integer n;
    begin
      @(negedge tx);#(CPB*10/2);
      if(tx!==0)$fatal(1,"bad start bit");
      for(n=0;n<8;n=n+1)begin #(CPB*10);b[n]=tx;end
      #(CPB*10);if(tx!==1)$fatal(1,"bad stop bit");
    end
  endtask

  // Count from the first WAITING edge, so the LAUNCH edge follows N+1
  // clocks for a programmed delay of N (including the zero-delay case).
  always @(posedge clk)begin
    cycle=cycle+1;
    if(!dut.rst)begin
      if(previous_rst || (dut.state==0 && previous_state!=0))wait_enter_cycle=cycle;
      if(dut.state==1)begin
        expected_delay=(launch_count==0)?START_CYCLES:(previous_fast?FAST_CYCLES:FRAME_CYCLES);
        if(cycle-wait_enter_cycle!=expected_delay+1)
          $fatal(1,"top output wait mismatch frame %0d: %0d expected %0d",
            launch_count,cycle-wait_enter_cycle,expected_delay+1);
        if(dut.startup_complete!==(launch_count!=0))$fatal(1,"startup_complete timing");
        previous_fast=dut.fast_frames;launch_count=launch_count+1;
      end
      if(dut.features_only && (dut.framebuffer_write!==0 || dut.framebuffer_read!==0))
        $fatal(1,"results-only mode accessed output framebuffer");
      if(dut.core_valid)begin
        active_frame=dut.frame_id-1;
        if(active_frame<0 || active_frame>4)$fatal(1,"unexpected core output frame");
        core_pixels[active_frame]=core_pixels[active_frame]+1;
      end
      if(dut.framebuffer_write)begin
        active_frame=dut.frame_id-1;
        if(dut.core_addr!==writes[active_frame])
          $fatal(1,"top output framebuffer rewrite address frame %0d",active_frame);
        reference_pixel=image_y(active_frame,(writes[active_frame]%OW)+1,(writes[active_frame]/OW)+1);
        if(dut.core_pixel!==reference_pixel[7:0])
          $fatal(1,"top output framebuffer pixel frame %0d address %0d",active_frame,writes[active_frame]);
        writes[active_frame]=writes[active_frame]+1;
      end
      if(dut.tx_start)begin
        active_frame=dut.frame_id-1;
        if(core_pixels[active_frame]!=N || writes[active_frame]!=((active_frame==2)?0:N))
          $fatal(1,"top output incomplete processing or framebuffer frame %0d",active_frame);
      end
    end
    previous_rst=dut.rst;previous_state=dut.state;
  end

  // Observe the storage itself as well as its enables. This catches an
  // implementation that accidentally leaves the RAM always block ungated.
  // The preceding packed frame has distinct first/last pixels, so an illicit
  // read when the result sender resets its address changes read_data.
  always @(negedge clk)begin
    if(!dut.rst && dut.features_only)begin
      if(!watching_features)begin
        for(snapshot_index=0;snapshot_index<N;snapshot_index=snapshot_index+1)
          frozen_framebuffer[snapshot_index]=dut.framebuffer[snapshot_index];
        frozen_read_data=dut.read_data;watching_features=1;
        if(frozen_framebuffer[0]===frozen_framebuffer[N-1])
          $fatal(1,"features read-gating fixture needs distinct first/last pixels");
      end else begin
        if(dut.read_data!==frozen_read_data)$fatal(1,"results-only mode changed RAM read register");
        for(snapshot_index=0;snapshot_index<N;snapshot_index=snapshot_index+1)
          if(dut.framebuffer[snapshot_index]!==frozen_framebuffer[snapshot_index])
            $fatal(1,"results-only mode changed framebuffer address %0d",snapshot_index);
      end
    end else watching_features=0;
  end

  // Every requested next mode is changed during acquisition, long before
  // the current UART packet, and must take effect only at the next frame.
  initial begin
    for(control_frame=0;control_frame<5;control_frame=control_frame+1)begin
      @(posedge dut.pattern_active);@(negedge clk);check_controls(control_frame);
      repeat(200)@(negedge clk);
      case(control_frame)
        0:sw=16'h1407; // Packed Sobel, dark target, fast subsequent frames.
        1:sw=16'h1cff; // Results-only wins over packing; bright target, threshold 960.
        2:sw=16'h100d; // Recover a complete grayscale v2 image.
        3:sw=16'h5c07; // Diagnostic page ignores all three output requests.
        4:sw=16'h8001; // Stop a sixth frame after the five-frame wire fixture.
      endcase
      repeat(10)@(negedge clk);check_controls(control_frame);
      wait(!dut.pattern_active);
    end
  end

  initial begin
    for(k=0;k<5;k=k+1)begin writes[k]=0;core_pixels[k]=0;end
    fd=$fopen("top_output_wire.bin","wb");
    if(fd==0)$fatal(1,"cannot open top_output_wire.bin");
    repeat(5)@(negedge clk);btnC=0;
    for(f=0;f<5;f=f+1)begin
      for(i=0;i<32;i=i+1)begin get_byte(received[i]);$fwrite(fd,"%c",received[i]);end
      header_bytes=(f==0)?32:72;
      payload_bytes=(f==2)?0:((f==1)?((N+7)/8):N);
      total=header_bytes+payload_bytes+2;
      for(i=32;i<total;i=i+1)begin get_byte(received[i]);$fwrite(fd,"%c",received[i]);end
      build_reference(f);
      if(u32(0)!==32'h31475046 || received[4]!==((f==0)?8'd1:((f==1 || f==2)?8'd3:8'd2)) ||
         received[5]!==((f==1 || f==2 || f==4)?8'd1:8'd0) ||
         u16(6)!==OW || u16(8)!==OH || u16(10)!==((f==2)?16'd960:16'd160) ||
         u32(12)!==f+1 || u32(16)!==payload_bytes)
        $fatal(1,"top output base header frame %0d",f);
      if(u32(20)!==ref_bright || u32(24)!==ref_bright_x || u32(28)!==ref_bright_y)
        $fatal(1,"top output bright statistics frame %0d",f);
      if(header_bytes==72)begin
        if(received[32]!==((f==1 || f==4)?8'd7:8'd5) || received[33]!==8'd96 ||
           u16(34)!==16'd1 || u16(36)!==16'd1 || u16(38)!==OW || u16(40)!==OH ||
           u16(42)!==ref_x_min || u16(44)!==ref_y_min || u16(46)!==ref_x_max || u16(48)!==ref_y_max ||
           u32(50)!==ref_area || u32(54)!==ref_x || u32(58)!==ref_y ||
           u32(62)!==MIN_AREA || u32(66)!==(N/2) ||
           received[70]!==((f==1)?8'd5:((f==2)?8'd6:8'd0)) || received[71]!==8'd0)
          $fatal(1,"top output target extension frame %0d",f);
      end
      for(i=0;i<payload_bytes;i=i+1)begin
        expected_byte=0;
        if(f==1)begin
          for(j=0;j<8;j=j+1)if(i*8+j<N)begin
            xx=(i*8+j)%OW+1;yy=(i*8+j)/OW+1;
            if(image_y(f,xx,yy)!=0)expected_byte=expected_byte|(1<<j);
          end
        end else begin xx=i%OW+1;yy=i/OW+1;expected_byte=image_y(f,xx,yy);end
        if(received[header_bytes+i]!==expected_byte[7:0])
          $fatal(1,"top output payload frame %0d byte %0d: %0d expected %0d",
            f,i,received[header_bytes+i],expected_byte);
      end
      crc=16'hffff;
      for(i=4;i<total-2;i=i+1)crc=update_crc(crc,received[i]);
      if(u16(total-2)!==crc)$fatal(1,"top output CRC frame %0d",f);
      check_controls(f);
      wait(dut.tx_done);@(negedge clk);
    end
    if(launch_count!=5 || writes[2]!=0 || core_pixels[2]!=N)
      $fatal(1,"top output frame count/results processing");
    // The diagnostic frame disables fast mode for its next wait, even when
    // SW12 remains a diagnostic request at its own frame boundary.
    if(dut.frame_wait_cycles!==FRAME_CYCLES)$fatal(1,"diagnostic page changed frame delay");
    $fclose(fd);
    $display("PASS top output five UART frames: legacy, packed, results, v2 raw, diagnostic; latching, delays, framebuffer gating and rewrite");
    $finish;
  end
  initial begin #5000000;$fatal(1,"top output timeout");end
endmodule

         u32(12)!==f+1 || u32(16)!==payload || received[70]!==transport || received[71]!==0)

      if(u16(total-2)!==crc)$fatal(1,"preview CRC");
      if(writes[f]!==((features)?0:N))$fatal(1,"preview incomplete framebuffer");
      wait(dut.tx_done);@(negedge clk);
      if(preview_enabled)countdown=(countdown+1)%INTERVAL;
    end
    if(RECORD)$fclose(fd);
    finished=1;
  end

    .cam_rrst_n(rrst),.cam_wen(wen),.uart_txd(tx),.led(led));

  defparam dut.config_camera.POWER_WAIT=3;
  defparam dut.config_camera.REG_WAIT=2;
  defparam dut.read_camera.QUARTER_CYCLES=2;

  // SCCB sink watches bus edges, including START/STOP. It only pulls the
  // ninth bit low; no sequencer state or internal ACK signal is overridden.
  reg camera_present=0,sink_ack=0,bus_active=0;
  integer bus_bit=0,bus_bytes=0,nack_transactions=0,ack_transactions=0;
  reg[7:0]bus_shift=0,bus_address=0;
  assign sda=sink_ack?1'b0:1'bz;
  always @(negedge sda)if(scl===1'b1)begin
    if(bus_active)$fatal(1,"unexpected SCCB START without STOP");
    bus_active=1;bus_bit=0;bus_bytes=0;bus_shift=0;bus_address=0;
  end
  always @(posedge scl)if(bus_active)begin
    if(bus_bit<8)begin
      bus_shift={bus_shift[6:0],sda};bus_bit=bus_bit+1;
      if(bus_bit==8)begin
        if(bus_bytes==0)bus_address=bus_shift;
        bus_bytes=bus_bytes+1;
      end
    end else bus_bit=9;
  end
  always @(negedge scl)begin
    if(bus_active && bus_bit==8)sink_ack=camera_present;
    else begin
      sink_ack=0;
      if(bus_bit==9)begin bus_bit=0;bus_shift=0;end
    end
  end
  always @(posedge sda)if(scl===1'b1 && bus_active)begin
    if(bus_address!==8'h42 && bus_address!==8'h43)
      $fatal(1,"bad SCCB device address %02x",bus_address);
    if(bus_address==8'h42 && bus_bytes==3)begin
      if(camera_present)ack_transactions=ack_transactions+1;
      else nack_transactions=nack_transactions+1;
    end
    bus_active=0;
  end

  // A camera writes UYVY bytes into a behavioral AL422 while WEN and HREF
  // are high. VSYNC drives its active-low write reset. RCLK independently
  // resets/advances the read pointer, presenting byte zero on its first edge.
  reg cam_pclk=0;reg[7:0]cam_write_data=0;

    end
  end
  always @(posedge rclk)begin
    if(!rrst)begin read_pointer=0;fifo_data<=0;end
    else begin
      fifo_data<=(read_pointer<W*H*2)?fifo_memory[read_pointer]:8'hde;
      read_pointer=read_pointer+1;
    end
  end

  integer retry_resets=0,capture_attempts=0,send_attempts=0,capture_sofs=0;
  integer capture_pixels=0,capture_frame_pixels=0,capture_completions=0;

  reg first_reset_released=0;

        if(dut.camera_fault!==1'b1)$fatal(1,"camera fault lost on retry reset");
      end

          $fatal(1,"physical camera reset pulse width");
        first_reset_released=1;last_release_cycle=cycle;
      end
      if(previous_config_reset && !dut.camera_config_rst &&
          cycle-last_release_cycle!=SETTLE_CYCLES)

          dut.camera_fault!==1'b1)

    previous_config_error=dut.config_error;

    end
  end

  endfunction
  function[15:0]u16(input integer offset);
    u16={received[offset+1],received[offset]};
  endfunction

      if(u16(32+N)!==crc)$fatal(1,"recovery UART CRC frame %0d",frame);
      packet_count=packet_count+1;
    end
    $fclose(fd);
  end

  initial begin
    repeat(5)@(negedge clk);btnC=0;
    wait(dut.pattern_active);repeat(200)@(negedge clk);sw=16'h8000;
    wait(packet_count==1);wait(dut.tx_done);@(negedge clk);
    if(nack_transactions<2 || retry_resets<2 || dut.config_done || capture_attempts!=0 ||
        dut.camera_fault!==1'b1 || led[15]!==1'b1)
      $fatal(1,"synthetic output did not survive repeated absent-camera NACK/retries");
    check_paused(0,1);
    // Reconnect only during physical reset; the complete register sequence
    // must succeed without touching the user reset button or internal state.
    @(negedge cam_reset);@(negedge clk);camera_present=1;
    if(dut.camera_fault!==1'b1 || led[15]!==1'b1)
      $fatal(1,"retry cleared the visible error before successful configuration");
    wait(dut.config_done);repeat(3)@(negedge clk);
    if(ack_transactions!=39 || dut.config_error || dut.camera_fault || led[15])
      $fatal(1,"camera did not recover all 39 configuration writes automatically");
    check_paused(0,1);

    wait(dut.cap_error);@(negedge clk);sw=16'h8000;

      $fatal(1,"missing frame end was not discarded cleanly");
    check_paused(2,3);

    // image memory and statistics, and pause/resume does not replay a frame.
    sw=0;supply_camera_frame(4);

module tb_transport;
  localparam W=4,H=2,CPB=4,N=32+W*H+2;

  always #5 clk=~clk;

    (.clk(clk),.rst(rst),.start(start),.edge_mode(mode),.threshold(threshold),
     .frame_id(frame_id),.bright_count(bc),.sum_x(sx),.sum_y(sy),
     .ram_addr(addr),.ram_data(rd),.busy(busy),.done(done),.uart_txd(tx));

  function [15:0] update_crc(input [15:0] c,input [7:0] b);

    integer k;

      @(negedge tx); #(CPB*10/2);

      #(CPB*10); if(tx!==1) $fatal(1,"bad stop bit");

      frame_id=7+f; mode=(f==0); threshold=180+f;
      @(negedge clk); start=1;
      @(negedge clk); start=0;
      // Mutate live metadata: transmitted values must remain latched.
      frame_id=999; threshold=999; mode=~mode;
      for(i=0;i<N;i=i+1) begin get_byte(received[i]); $fwrite(fd,"%c",received[i]); end

      if(received[20]!=2 || received[24]!=7 || received[28]!=5) $fatal(1,"statistics");

  end
  initial begin #1000000; $fatal(1,"timeout"); end
endmodule
